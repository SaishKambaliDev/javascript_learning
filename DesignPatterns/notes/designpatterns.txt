is pattern ka main idea hai object creationn ko ek function ke through control karna

har baar jab tum factory function call karte ho, tumhe ek naya object milta hai jisme apnne methods aur (agar chaho to ) private data ho sakta hai

function createProduct(name, price){
    let stock = 10;
    return {
        name,
        price,
        checkStock() {
            console.log(stock);
        },
        buy(qty){
            if(qty <= stock)
            {
                stock -= qty;
                console.log(`${qty} pieces booked - ${stock} pieces left.`);
            }
            else{
                console.error(`we only have ${stock} pieces left.`);
            }
        },
        refill(qty){
            stock += qty;
            console.log(`refilled the stock - ${stock} pieces now.`);
        },
    };
}

let phone = createProduct("phone", 700000);
phone.buy(6);

debouncing -> aap koi action kar rahe ho and aap ye nahi chaate har action pe kuch ho, jab bhi mere actions ke beech mei koi specific gap aajaye to fir action perform ho


let input = document.querySelector("input");

function debounce(fnc, delay){
    let timer;
    return function(...args){
        clearTimeout(timer);
        timer = setTimeout(() => {
            fnc(...args);
        }, delay);
    };
}

input.addEventListener("input", debounce(function () {
    console.log("ran");
}, 1000)
);
